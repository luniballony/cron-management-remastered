# BUILT WITH:
Backend:
- node.js
- typescript.js
- express.js
- postgreSQL
- jest

Frontend:
- react
- typeScript
- TailWind

# INSTALATION:
simply run:
- npm install

or install packages seperately: 
backend:
- npm install express cors dotenv pg
- npm install node-cron
- npm install -D typescript ts-node-dev
- npm install -D @types/node
- npm install -D @types/express
- npm install -D @types/cors

# TO RUN: 
backend:
- cd /backend/src
- npm run dev
